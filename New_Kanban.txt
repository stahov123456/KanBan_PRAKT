<script>
  let cardCounter = Date.now();

  // Тема
  const toggle = document.getElementById('themeToggle');
  const savedTheme = localStorage.getItem('theme') || 'light';
  if (savedTheme === 'dark') {
    document.body.classList.add('dark');
    toggle.textContent = '☀️';
  }

  toggle.onclick = () => {
    document.body.classList.toggle('dark');
    const isDark = document.body.classList.contains('dark');
    toggle.textContent = isDark ? '☀️' : '🌙';
    localStorage.setItem('theme', isDark ? 'dark' : 'light');
  };

  // Добавление карточки
  function addCard(btn) {
    const list = btn.previousElementSibling;
    const columnId = btn.closest('.column').dataset.columnId;
    const text = prompt('Текст карточки:')?.trim();
    if (!text) return;

    const card = document.createElement('div');
    card.className = 'main-card';
    card.dataset.cardId = 'c' + cardCounter++;
    card.innerHTML = `
      ${text}
      <button class="delete-btn" onclick="this.parentElement.remove(); saveBoardState();">×</button>
    `;

    if (columnId === 'in-progress') card.classList.add('blue');

    list.appendChild(card);
    saveBoardState();
  }

  // Сохранение
  function saveBoardState() {
    const state = {};
    document.querySelectorAll('.column').forEach(col => {
      const id = col.dataset.columnId;
      state[id] = [];
      col.querySelectorAll('.main-card').forEach(c => {
        state[id].push(c.outerHTML);
      });
    });
    localStorage.setItem('kanbanState', JSON.stringify(state));
  }

  // Загрузка
  function loadBoardState() {
    const data = localStorage.getItem('kanbanState');
    if (!data) return;

    try {
      const state = JSON.parse(data);
      Object.entries(state).forEach(([colId, cardsHtml]) => {
        const list = document.querySelector(`[data-column-id="${colId}"] .cards-list`);
        if (!list) return;
        list.innerHTML = ''; // очищаем только список карточек
        cardsHtml.forEach(html => {
          const div = document.createElement('div');
          div.innerHTML = html;
          list.appendChild(div.firstChild);
        });
      });
    } catch (e) {
      console.warn('Состояние повреждено → сброс', e);
      localStorage.removeItem('kanbanState');
    }
  }

  // Инициализация drag-and-drop
  function initSortable() {
    document.querySelectorAll('.cards-list').forEach(list => {
      new Sortable(list, {
        group: 'kanban',
        animation: 140,
        draggable: '.main-card',
        ghostClass: 'sortable-ghost',
        chosenClass: 'sortable-chosen',
        onEnd: evt => {
          const card = evt.item;
          const newColumn = evt.to.closest('.column');
          if (!newColumn) return;

          const newId = newColumn.dataset.columnId;

          if (newId === 'in-progress') {
            card.classList.add('blue');
            card.querySelector('.tag')?.classList.add('blue');
          } else {
            card.classList.remove('blue');
            card.querySelector('.tag')?.classList.remove('blue');
          }

          saveBoardState();
        }
      });
    });
  }

  // Старт
  window.addEventListener('load', () => {
    loadBoardState();
    initSortable();
  });
</script>
